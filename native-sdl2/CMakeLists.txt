cmake_minimum_required(VERSION 3.21)
project(godog LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(SDL2 CONFIG REQUIRED)
find_package(ZLIB REQUIRED)
add_library(godog_engine STATIC src/engine/core.cpp src/demo.cpp src/surface_platform.cpp src/assets/original_asset_loader.cpp)
target_link_libraries(godog_engine PUBLIC ZLIB::ZLIB)
target_include_directories(godog_engine PUBLIC src)
if(MSVC)
  target_compile_options(godog_engine PRIVATE /W4 /fp:strict)
  target_compile_options(godog_engine PUBLIC /utf-8)
else()
  target_compile_options(godog_engine PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -Wno-unused-parameter -Wno-unused-variable)
  if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    target_compile_options(godog_engine PRIVATE -Wno-parentheses-equality)
  endif()
endif()
add_executable(godog src/main.cpp)
target_compile_definitions(godog PRIVATE SDL_MAIN_HANDLED)
target_link_libraries(godog PRIVATE godog_engine SDL2::SDL2)
add_custom_command(TARGET godog POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/../java-desktop/assets" "$<TARGET_FILE_DIR:godog>/assets")
install(TARGETS godog RUNTIME DESTINATION .)
install(DIRECTORY ../java-desktop/assets DESTINATION .)
include(CTest)
option(GODOG_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
if(GODOG_SANITIZE AND NOT MSVC)
  target_compile_options(godog_engine PUBLIC -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  target_link_options(godog_engine PUBLIC -fsanitize=address,undefined)
endif()
set(GODOG_REFERENCE_DIR "" CACHE PATH "Fixtures produced by tools/ExportReference.java")
if(BUILD_TESTING)
  add_executable(godog_semantics_test tests/semantics.cpp)
  target_link_libraries(godog_semantics_test PRIVATE godog_engine)
  add_test(NAME java_semantics_and_lifecycle COMMAND godog_semantics_test "${CMAKE_CURRENT_SOURCE_DIR}/../java-desktop/assets")
  add_executable(godog_playback_test tests/playback.cpp)
  target_link_libraries(godog_playback_test PRIVATE godog_engine)
  add_test(NAME complete_playback COMMAND godog_playback_test "${CMAKE_CURRENT_SOURCE_DIR}/../java-desktop/assets")
  set_tests_properties(complete_playback PROPERTIES TIMEOUT 300)
  add_executable(godog_reference_test tests/reference.cpp)
  target_link_libraries(godog_reference_test PRIVATE godog_engine)
  add_executable(godog_audio_test tests/audio.cpp)
  target_link_libraries(godog_audio_test PRIVATE godog_engine)
  if(GODOG_REFERENCE_DIR)
    add_test(NAME java_image_and_raster_reference COMMAND godog_reference_test "${CMAKE_CURRENT_SOURCE_DIR}/../java-desktop/assets" "${GODOG_REFERENCE_DIR}")
    add_test(NAME java_xm_reference COMMAND godog_audio_test "${CMAKE_CURRENT_SOURCE_DIR}/../java-desktop/assets/data/rocket.xm" "${GODOG_REFERENCE_DIR}/music.pcm")
  endif()
endif()
if(WIN32)
  # vcpkg normally deploys these too; explicitly retain the runtime DLLs for
  # non-vcpkg multi-config builds and relocatable installations.
  add_custom_command(TARGET godog POST_BUILD
    COMMAND ${CMAKE_COMMAND} "-DLIBRARIES=$<TARGET_RUNTIME_DLLS:godog>" "-DDESTINATION=$<TARGET_FILE_DIR:godog>" -P "${CMAKE_CURRENT_SOURCE_DIR}/tools/copy_runtime.cmake")
  install(FILES $<TARGET_RUNTIME_DLLS:godog> DESTINATION .)
endif()
